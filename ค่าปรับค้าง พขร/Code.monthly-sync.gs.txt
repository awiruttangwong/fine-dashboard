function onOpen() {
  try {
    SpreadsheetApp.getUi()
      .createMenu(SHIPPING_CONFIG.menuName)
      .addItem('ซิงก์ข้อมูล', 'syncShippingData')
      .addSeparator()
      .addItem('รวมข้อมูล', 'consolidateShippingDataV4')
      .addToUi();
  } catch (error) {
    console.info('[shipping-onOpen] ข้ามการสร้างเมนู เพราะ context นี้ไม่มี Spreadsheet UI: ' + error.message);
  }
}

// ใช้งานประจำ: อัปเดตข้อมูลจากต้นทางลง SUM
function syncShippingData() {
  runShippingOperation_(true);
}

// ใช้เมื่อต้องการสร้าง SUM ใหม่: คงชื่อฟังก์ชันเดิมไว้
function consolidateShippingDataV4() {
  runShippingOperation_(false);
}

function runShippingOperation_(preserveSumState) {
  var lock = LockService.getDocumentLock();

  try {
    lock.waitLock(30000);

    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sumSheet = getOrCreateSheet_(ss, SHIPPING_CONFIG.sumSheetName);
    var sourceResult = collectSourceData_(ss);

    // ป้องกันข้อมูลปลายทางสูญหายเมื่อชื่อชีทผิด ต้นทางว่าง หรืออ่านข้อมูลไม่สำเร็จ
    if (sourceResult.rows.length === 0) {
      throw new Error(buildNoDataMessage_(sourceResult));
    }

    var finalData = sourceResult.rows.map(function(row) {
      return row.slice(0, SHIPPING_CONFIG.headers.length);
    });

    sortByRecipient_(finalData);
    writeSumSheet_(sumSheet, finalData);
    writeStatusSheets_(ss, preserveSumState ? sourceResult.statusRows : createStatusBuckets_());
    ensureStatusSheetOrder_(ss);

    var modeText = preserveSumState ? 'ซิงก์ข้อมูลสำเร็จ' : 'รวมข้อมูลใหม่สำเร็จ';
    var missingText = sourceResult.missingSheets.length > 0
      ? '\nไม่พบชีท: ' + sourceResult.missingSheets.join(', ')
      : '';
    var statusText = preserveSumState
      ? '\nรอปรับ ' + sourceResult.statusRows['รอปรับ'].length +
        ', ปรับได้ ' + sourceResult.statusRows['ปรับได้'].length +
        ', ปรับไม่ได้ ' + sourceResult.statusRows['ปรับไม่ได้'].length
      : '\nสร้างชีท รอปรับ, ปรับได้, ปรับไม่ได้ แล้ว';
    var unmatchedText = sourceResult.unmatchedStatusCount > 0
      ? '\nยังไม่เข้าเงื่อนไขสถานะ ' + sourceResult.unmatchedStatusCount + ' รายการ'
      : '';
    var skippedText = buildSkippedSourceText_(sourceResult);
    notifyUser_(modeText + '\nจำนวน ' + finalData.length + ' รายการ' + statusText + unmatchedText + skippedText + missingText);
  } catch (error) {
    console.error('[shipping-sync] ' + error.stack);
    notifyUser_('ดำเนินการไม่สำเร็จ\n' + error.message);
  } finally {
    if (lock.hasLock()) {
      lock.releaseLock();
    }
  }
}

function onEdit(e) {
  return;

function collectSourceData_(ss) {
  var rows = [];
  var statusRows = createStatusBuckets_();
  var missingSheets = [];
  var shortSheets = [];
  var skippedRequiredCount = 0;
  var skippedDateLikeNameCount = 0;
  var skippedInvalidRecordCount = 0;
  var unmatchedStatusCount = 0;
  var monthHint = resolveWorkbookMonthHint_(ss);

  SHIPPING_CONFIG.sources.forEach(function(config) {
    var sheet = ss.getSheetByName(config.name);
    if (!sheet) {
      missingSheets.push(config.name);
      return;
    }

    var lastRow = sheet.getLastRow();
    var lastColumn = sheet.getLastColumn();
    var maxRequiredColumn = config.requiredColIdx + 1;
    var hasStatusColumn = lastColumn >= (config.statusColIdx + 1);
    if (lastRow <= 1) return;
    if (lastColumn < maxRequiredColumn) {
      shortSheets.push(config.name + ' ต้องมีถึงคอลัมน์ ' + columnLetter_(maxRequiredColumn) + ' เพื่ออ่านยอดปรับ');
      return;
    }

    var range = sheet.getRange(2, 1, lastRow - 1, lastColumn);
    var values = range.getValues();
    var displayValues = range.getDisplayValues();
    var sheetMonthHint = monthHint || inferMonthHintFromRows_(values, displayValues, config.cols[0]);
    values.forEach(function(sourceRow, rowIndex) {
      var displayRow = displayValues[rowIndex] || [];
      var fineAmount = parseFineAmount_(sourceRow[config.requiredColIdx], displayRow[config.requiredColIdx]);
      if (fineAmount === null || fineAmount <= 0) {
        skippedRequiredCount++;
        return;
      }

      var row = config.cols.map(function(columnIndex, outputIndex) {
        var value = sourceRow[columnIndex] !== undefined ? sourceRow[columnIndex] : '';
        var displayValue = displayRow[columnIndex] !== undefined ? displayRow[columnIndex] : '';
        return outputIndex === 0 ? parseCustomDate(value, displayValue, sheetMonthHint) : value;
      });

      if (!hasMeaningfulRecord_(row)) {
        skippedInvalidRecordCount++;
        return;
      }

      if (hasDateLikeNameFields_(row)) {
        skippedDateLikeNameCount++;
        return;
      }

      var status = resolveFineStatus_(
        sourceRow[10],
        displayRow[10],
        sourceRow[14],
        displayRow[14],
        sourceRow[15],
        displayRow[15],
        hasStatusColumn ? sourceRow[config.statusColIdx] : '',
        hasStatusColumn ? displayRow[config.statusColIdx] : ''
      );
      rows.push(row);
      if (status) {
        statusRows[status].push(row.slice(0, SHIPPING_CONFIG.headers.length));
      } else {
        unmatchedStatusCount++;
      }
    });
  });

  return {
    rows: rows,
    missingSheets: missingSheets,
    shortSheets: shortSheets,
    skippedRequiredCount: skippedRequiredCount,
    skippedDateLikeNameCount: skippedDateLikeNameCount,
    skippedInvalidRecordCount: skippedInvalidRecordCount,
    statusRows: statusRows,
    unmatchedStatusCount: unmatchedStatusCount
  };
}

function buildNoDataMessage_(sourceResult) {
  var parts = ['ไม่พบข้อมูลต้นทางที่พร้อมซิงก์ ระบบจึงไม่ได้แก้ไข SUM'];
  var diagnosticText = buildSourceDiagnosticsText_(sourceResult);
  if (diagnosticText) parts.push(diagnosticText);
  return parts.join('\n');
}

function buildSkippedSourceText_(sourceResult) {
  var parts = [];
  if (sourceResult.shortSheets.length > 0) {
    parts.push('ชีทคอลัมน์ไม่ครบ: ' + sourceResult.shortSheets.join(', '));
  }
  if (sourceResult.skippedRequiredCount > 0) {
    parts.push('ข้ามแถว K ว่างหรือไม่ใช่ตัวเลขมากกว่า 0 จำนวน ' + sourceResult.skippedRequiredCount + ' รายการ');
  }
  if (sourceResult.skippedDateLikeNameCount > 0) {
    parts.push('ข้ามแถวชื่อผิดรูปแบบเป็นวันที่ ' + sourceResult.skippedDateLikeNameCount + ' รายการ');
  }
  if (sourceResult.skippedInvalidRecordCount > 0) {
    parts.push('ข้ามแถวข้อมูลหลักไม่ครบ ' + sourceResult.skippedInvalidRecordCount + ' รายการ');
  }
  return parts.length > 0 ? '\n' + parts.join('\n') : '';
}

function buildSourceDiagnosticsText_(sourceResult) {
  var parts = [];
  if (sourceResult.missingSheets.length > 0) {
    parts.push('ไม่พบชีท: ' + sourceResult.missingSheets.join(', '));
  }
  if (sourceResult.shortSheets.length > 0) {
    parts.push('ชีทคอลัมน์ไม่ครบ: ' + sourceResult.shortSheets.join(', '));
  }
  if (sourceResult.skippedRequiredCount > 0) {
    parts.push('ข้ามแถว K ว่างหรือไม่ใช่ตัวเลขมากกว่า 0 จำนวน ' + sourceResult.skippedRequiredCount + ' รายการ');
  }
  if (sourceResult.skippedDateLikeNameCount > 0) {
    parts.push('ข้ามแถวชื่อผิดรูปแบบเป็นวันที่ ' + sourceResult.skippedDateLikeNameCount + ' รายการ');
  }
  if (sourceResult.skippedInvalidRecordCount > 0) {
    parts.push('ข้ามแถวข้อมูลหลักไม่ครบ ' + sourceResult.skippedInvalidRecordCount + ' รายการ');
  }
  return parts.join('\n');
}

function hasMeaningfulRecord_(row) {
  // ต้องมีวันที่ ลูกค้า บาร์โค้ด หรือเส้นทางอย่างน้อยหนึ่งค่า และมียอดปรับ
  var hasIdentity = [row[0], row[1], row[2], row[3]].some(function(value) {
    return value !== '' && value !== null && value !== undefined;
  });
  return hasIdentity && row[6] !== '' && row[6] !== null && row[6] !== undefined;
}

function keyPart_(value) {
  if (value === null || value === undefined) return '';
  return String(value).trim().replace(/\s+/g, ' ').toUpperCase();
}

function sortByRecipient_(rows) {
  var counts = {};
  rows.forEach(function(row) {
    var recipient = keyPart_(row[5]);
    if (recipient) counts[recipient] = (counts[recipient] || 0) + 1;
  });

  rows.sort(function(a, b) {
    var recipientA = keyPart_(a[5]);
    var recipientB = keyPart_(b[5]);
    if (!recipientA && recipientB) return 1;
    if (recipientA && !recipientB) return -1;
    if (!recipientA && !recipientB) return 0;

    var countDifference = (counts[recipientB] || 0) - (counts[recipientA] || 0);
    return countDifference || recipientA.localeCompare(recipientB, 'th');
  });
}

function writeSumSheet_(sheet, rows) {
  var sumColumnCount = SHIPPING_CONFIG.headers.length;
  ensureSheetSize_(sheet, rows.length + 1, sumColumnCount);
  clearBody_(sheet, sumColumnCount);
  sheet.getRange(1, 1, 1, sumColumnCount).setValues([SHIPPING_CONFIG.headers]);
  formatHeader_(sheet, sumColumnCount);

  SHIPPING_CONFIG.columnWidths.forEach(function(width, index) {
    sheet.setColumnWidth(index + 1, width);
  });

  clearUnusedManagedColumns_(sheet, sumColumnCount, 1);

  if (rows.length === 0) return;

  sheet.getRange(2, 1, rows.length, 1).setNumberFormat('@');
  sheet.getRange(2, 1, rows.length, sumColumnCount).setValues(rows);
  sheet.getRange(2, 7, rows.length, 3).setNumberFormat('#,##0.00');
  sheet.getRange(2, 1, rows.length, sumColumnCount).setWrap(false).setVerticalAlignment('middle');
  sheet.setRowHeights(2, rows.length, 25);
}

function writeStatusSheets_(ss, statusRows) {
  SHIPPING_CONFIG.statusSheets.forEach(function(config) {
    var rows = (statusRows && statusRows[config.name] ? statusRows[config.name] : []).slice();
    sortByRecipient_(rows);
    writeStatusSheet_(getOrCreateSheet_(ss, config.name), rows);
  });
}

function ensureStatusSheetOrder_(ss) {
  var sumSheet = ss.getSheetByName(SHIPPING_CONFIG.sumSheetName);
  if (!sumSheet) return;

  var activeSheet = ss.getActiveSheet();
  var targetIndex = sumSheet.getIndex();

  SHIPPING_CONFIG.statusSheets.forEach(function(config) {
    var sheet = ss.getSheetByName(config.name);
    if (!sheet) return;
    targetIndex++;
    moveSheetToIndex_(ss, sheet, targetIndex);
  });

  if (activeSheet) {
    ss.setActiveSheet(activeSheet);
  }
}

function writeStatusSheet_(sheet, rows) {
  var columnCount = SHIPPING_CONFIG.headers.length;
  ensureSheetSize_(sheet, Math.max(rows.length + 1, 2), columnCount);
  clearBody_(sheet, columnCount);
  sheet.getRange(1, 1, 1, columnCount).setValues([SHIPPING_CONFIG.headers]);
  formatHeader_(sheet, columnCount);

  SHIPPING_CONFIG.columnWidths.forEach(function(width, index) {
    sheet.setColumnWidth(index + 1, width);
  });

  clearUnusedManagedColumns_(sheet, columnCount, 1);

  if (rows.length === 0) return;

  sheet.getRange(2, 1, rows.length, 1).setNumberFormat('@');
  sheet.getRange(2, 1, rows.length, columnCount).setValues(rows);
  sheet.getRange(2, 7, rows.length, 3).setNumberFormat('#,##0.00');
  sheet.getRange(2, 1, rows.length, columnCount).setWrap(false).setVerticalAlignment('middle');
  sheet.setRowHeights(2, rows.length, 25);
}

function createStatusBuckets_() {
  var buckets = {};
  SHIPPING_CONFIG.statusSheets.forEach(function(config) {
    buckets[config.name] = [];
  });
  return buckets;
}

function resolveFineStatus_(fineValue, fineDisplay, paidValue, paidDisplay, remainingValue, remainingDisplay, closeValue, closeDisplay) {
  var closeText = String(closeDisplay || closeValue || '').trim();
  if (closeText.indexOf('จบจ่าย') !== -1) return 'ปรับไม่ได้';

  var fine = parseFineAmount_(fineValue, fineDisplay);
  var paid = parseFineAmount_(paidValue, paidDisplay);
  var remaining = parseFineAmount_(remainingValue, remainingDisplay);
  if (fine === null || fine <= 0) return '';

  if (isZeroOrBlankAmount_(paid) && remaining !== null && amountsEqual_(remaining, fine * -1)) {
    return 'รอปรับ';
  }

  if (paid !== null && paid > 0 && amountsEqual_(paid, fine) && isZeroOrBlankAmount_(remaining)) {
    return 'ปรับได้';
  }

  return '';
}

function parseFineAmount_(value, displayValue) {
  var displayText = String(displayValue === null || displayValue === undefined ? '' : displayValue).trim();
  if (displayText !== '') return parseFineAmountText_(displayText);
  if (typeof value === 'number') return isFinite(value) ? value : null;
  return parseFineAmountText_(value);
}

function parseFineAmountText_(value) {
  if (value === '' || value === null || value === undefined) return null;
  var normalized = String(value)
    .trim()
    .replace(/,/g, '')
    .replace(/[^0-9.\-]/g, '');
  if (normalized === '' || normalized === '-' || normalized === '.' || normalized === '-.') return null;

  var parsed = Number(normalized);
  return isFinite(parsed) ? parsed : null;
}

function isZeroOrBlankAmount_(value) {
  return value === null || amountsEqual_(value, 0);
}

function amountsEqual_(left, right) {
  return Math.abs(Number(left) - Number(right)) < 0.005;
}

function hasDateLikeNameFields_(row) {
  return looksLikeDateText_(row[4]) || looksLikeDateText_(row[5]);
}

function matchesTargetMonth_(value, targetMonth) {
  var month = extractMonthNumber_(value);
  return month === null || month === targetMonth;
}

function extractMonthNumber_(value) {
  if (value === '' || value === null || value === undefined) return null;
  if (value instanceof Date && !isNaN(value.getTime())) return value.getMonth() + 1;

  var text = String(value).trim();
  var parts = text.split(/[-/]/);
  if (parts.length !== 3) return null;

  if (String(parts[0]).length === 4) {
    var isoMonth = parseInt(parts[1], 10);
    return isFinite(isoMonth) ? isoMonth : null;
  }

  var displayMonth = parseInt(parts[1], 10);
  return isFinite(displayMonth) ? displayMonth : null;
}

function formatHeader_(sheet, columnCount) {
  sheet.getRange(1, 1, 1, columnCount)
    .setFontSize(11)
    .setFontWeight('bold')
    .setVerticalAlignment('middle')
    .setHorizontalAlignment('center')
    .setWrap(false);
  sheet.setRowHeight(1, 30);
}

function clearBody_(sheet, columnCount) {
  var rowCount = Math.max(sheet.getLastRow() - 1, 0);
  if (rowCount === 0) return;
  var range = sheet.getRange(2, 1, rowCount, columnCount);
  range.clearContent();
  range.clearDataValidations();
}

function clearUnusedManagedColumns_(sheet, keepColumnCount, extraManagedColumnCount) {
  if (extraManagedColumnCount <= 0) return;
  var rowCount = Math.max(sheet.getLastRow(), 1);
  var range = sheet.getRange(1, keepColumnCount + 1, rowCount, extraManagedColumnCount);
  range.clearContent();
  range.clearFormat();
  range.clearDataValidations();
}

function columnLetter_(columnNumber) {
  var letter = '';
  while (columnNumber > 0) {
    var remainder = (columnNumber - 1) % 26;
    letter = String.fromCharCode(65 + remainder) + letter;
    columnNumber = Math.floor((columnNumber - 1) / 26);
  }
  return letter;
}

function ensureSheetSize_(sheet, requiredRows, requiredColumns) {
  if (sheet.getMaxRows() < requiredRows) {
    sheet.insertRowsAfter(sheet.getMaxRows(), requiredRows - sheet.getMaxRows());
  }
  if (sheet.getMaxColumns() < requiredColumns) {
    sheet.insertColumnsAfter(sheet.getMaxColumns(), requiredColumns - sheet.getMaxColumns());
  }
}

function getOrCreateSheet_(ss, name) {
  return ss.getSheetByName(name) || ss.insertSheet(name);
}

function moveSheetToIndex_(ss, sheet, targetIndex) {
  if (!sheet || sheet.getIndex() === targetIndex) return;
  sheet.activate();
  ss.moveActiveSheet(targetIndex);
}

function notifyUser_(message) {
  try {
    SpreadsheetApp.getUi().alert(message);
  } catch (error) {
    console.info('[shipping-notify] ' + message);
  }
}

function resolveWorkbookMonthHint_(ss) {
  var text = ss && ss.getName ? String(ss.getName()) : '';
  var rawText = text;
  var upperText = rawText.toUpperCase();
  var labelMatch = upperText.match(/\bM(1[0-2]|[1-9])\b/);
  if (labelMatch) return Number(labelMatch[1]);

  var explicitMonthPatterns = [
    /\b(?:20\d{2})[\/\-_ ](0?[1-9]|1[0-2])\b/,
    /\b(0?[1-9]|1[0-2])[\/\-_ ](?:20\d{2})\b/,
    /(?:เดือน|MONTH)\s*(0?[1-9]|1[0-2])\b/i
  ];
  for (var i = 0; i < explicitMonthPatterns.length; i++) {
    var patternMatch = rawText.match(explicitMonthPatterns[i]);
    if (patternMatch) return Number(patternMatch[1]);
  }

  var monthNamePatterns = [
    { month: 1, regex: /(JANUARY|JAN|มกราคม|ม\.ค\.?|มค)/i },
    { month: 2, regex: /(FEBRUARY|FEB|กุมภาพันธ์|ก\.พ\.?|กพ)/i },
    { month: 3, regex: /(MARCH|MAR|มีนาคม|มี\.ค\.?|มีค)/i },
    { month: 4, regex: /(APRIL|APR|เมษายน|เม\.ย\.?|เมย)/i },
    { month: 5, regex: /(MAY|พฤษภาคม|พ\.ค\.?|พค)/i },
    { month: 6, regex: /(JUNE|JUN|มิถุนายน|มิ\.ย\.?|มิย)/i },
    { month: 7, regex: /(JULY|JUL|กรกฎาคม|ก\.ค\.?|กค)/i },
    { month: 8, regex: /(AUGUST|AUG|สิงหาคม|ส\.ค\.?|สค)/i },
    { month: 9, regex: /(SEPTEMBER|SEPT|SEP|กันยายน|ก\.ย\.?|กย)/i },
    { month: 10, regex: /(OCTOBER|OCT|ตุลาคม|ต\.ค\.?|ตค)/i },
    { month: 11, regex: /(NOVEMBER|NOV|พฤศจิกายน|พ\.ย\.?|พย)/i },
    { month: 12, regex: /(DECEMBER|DEC|ธันวาคม|ธ\.ค\.?|ธค)/i }
  ];
  for (var monthIndex = 0; monthIndex < monthNamePatterns.length; monthIndex++) {
    if (monthNamePatterns[monthIndex].regex.test(rawText)) {
      return monthNamePatterns[monthIndex].month;
    }
  }

  return null;
}

function inferMonthHintFromRows_(values, displayValues, dateColumnIndex) {
  var firstCounts = {};
  var secondCounts = {};
  var candidateCount = 0;

  values.forEach(function(row, rowIndex) {
    var displayRow = displayValues[rowIndex] || [];
    var text = String(displayRow[dateColumnIndex] || row[dateColumnIndex] || '').trim();
    var match = text.match(/^(\d{1,2})[\/\-](\d{1,2})[\/\-]\d{2,4}$/);
    if (!match) return;

    var first = Number(match[1]);
    var second = Number(match[2]);
    if (!first || !second || first > 12 || second > 12) return;

    candidateCount++;
    firstCounts[first] = (firstCounts[first] || 0) + 1;
    secondCounts[second] = (secondCounts[second] || 0) + 1;
  });

  if (candidateCount < 3) return null;

  var firstBest = bestCount_(firstCounts);
  var secondBest = bestCount_(secondCounts);
  var threshold = Math.max(3, Math.ceil(candidateCount * 0.6));

  if (firstBest.count >= threshold && firstBest.count > secondBest.count) return firstBest.value;
  if (secondBest.count >= threshold && secondBest.count > firstBest.count) return secondBest.value;
  return null;
}

function bestCount_(counts) {
  return Object.keys(counts).reduce(function(best, key) {
    var count = counts[key];
    if (count > best.count) {
      return { value: Number(key), count: count };
    }
    return best;
  }, { value: null, count: 0 });
}

function looksLikeDateText_(value) {
  if (value === '' || value === null || value === undefined) return false;
  return /^\d{1,2}[\/-]\d{1,2}[\/-]\d{2,4}$/.test(String(value).trim()) ||
    /^\d{4}[\/-]\d{1,2}[\/-]\d{1,2}$/.test(String(value).trim());
}

function parseCustomDate(value, displayValue, monthHint) {
  if (value === '' || value === null || value === undefined) return '';

  if (Object.prototype.toString.call(value) === '[object Date]' && !isNaN(value.getTime())) {
    return formatDateText_(value.getFullYear(), value.getMonth() + 1, value.getDate());
  }

  var displayText = String(displayValue === null || displayValue === undefined ? '' : displayValue).trim();
  var text = displayText || String(value).trim();
  var parsedTextDate = parseDateTextParts_(text, monthHint);
  if (parsedTextDate) return formatDateText_(parsedTextDate.year, parsedTextDate.month, parsedTextDate.day);

  return value;
}

function parseDateTextParts_(text, monthHint) {
  if (!text) return null;
  var parts = text.split(/[-/]/);
  if (parts.length !== 3) return null;

  var first = parseInt(parts[0], 10);
  var second = parseInt(parts[1], 10);
  var third = parseInt(parts[2], 10);
  if (!isFinite(first) || !isFinite(second) || !isFinite(third)) return null;

  var year;
  var month;
  var day;

  if (String(parts[0]).length === 4) {
    year = first;
    month = second;
    day = third;
  } else {
    year = third;

    if (first > 12 && second <= 12) {
      day = first;
      month = second;
    } else if (second > 12 && first <= 12) {
      month = first;
      day = second;
    } else if (SHIPPING_CONFIG.preferTargetMonthOnAmbiguousDates && monthHint && first === monthHint && second !== monthHint) {
      month = first;
      day = second;
    } else if (SHIPPING_CONFIG.preferTargetMonthOnAmbiguousDates && monthHint && second === monthHint && first !== monthHint) {
      day = first;
      month = second;
    } else {
      // ถ้าไม่มีเดือนเป้าหมายช่วย ระบบปลายทางใช้วัน/เดือน/ปี เป็นค่าเริ่มต้น
      day = first;
      month = second;
    }
  }

  if (year < 100) year += 2000;

  var parsed = new Date(year, month - 1, day);
  if (parsed.getFullYear() !== year || parsed.getMonth() !== month - 1 || parsed.getDate() !== day) {
    return null;
  }

  return { year: year, month: month, day: day };
}

function formatDateText_(year, month, day) {
  return [
    String(day).padStart(2, '0'),
    String(month).padStart(2, '0'),
    String(year)
  ].join('/');
}
